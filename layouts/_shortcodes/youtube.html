{{- /* A YouTube poster that links out to youtube.com — no iframe, no cookies. {{< youtube qtIqKaDlqXo >}} {{< youtube "qtIqKaDlqXo" "/uploads/youtubeposter.jpg" >}} {{< youtube id="qtIqKaDlqXo" image="/uploads/youtubeposter.jpg" >}} Both calling conventions are supported: the Hugobricks one, where the poster is a local image, and the hugocodex.org one, where only the id is given and the poster comes from YouTube's thumbnail server at build time (maxresdefault first — not every video has one — then sddefault). A remote fetch that fails only costs the poster, never the build. The play button and wordmark are painted by CSS over the link — style.css, "video shortcodes" — so the markup is nothing but an anchor with an image. @param {string} [0|id] the YouTube video id @param {string} [1|image] optional local poster image; omit it to fetch the thumbnail from YouTube at build time */ -}} {{- $id := "" -}} {{- $image := "" -}} {{- if .IsNamedParams -}} {{- $id = .Get "id" -}} {{- $image = .Get "image" -}} {{- else -}} {{- $id = .Get 0 -}} {{- $image = .Get 1 -}} {{- end -}} {{- /* The poster: a processed resource when one can be had, the raw path otherwise, an empty src only when a remote-only call is offline. */ -}} {{- $poster := "" -}} {{- $src := $image -}} {{- if $image -}} {{- $poster = resources.GetMatch $image -}} {{- else -}} {{- range slice "maxresdefault" "sddefault" -}} {{- if not $poster -}} {{- with try (resources.GetRemote (printf "https://i.ytimg.com/vi/%s/%s.jpg" $id .)) -}} {{- with .Err -}} {{- warnf "youtube shortcode: %s" . -}} {{- else -}} {{- $poster = .Value -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $poster -}} {{- $src = (.Fill "1200x675 webp Center q50").RelPermalink -}} {{- end -}}
