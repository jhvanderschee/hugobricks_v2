{{- /* A Vimeo poster that links out to vimeo.com — no iframe, no cookies. {{< vimeo 727145223 >}} {{< vimeo id="727145223" image="/uploads/gallery/01.jpg" >}} Both calling conventions are supported: the Hugobricks one, where the poster is a local image, and the hugocodex.org one, where only the id is given and the poster is looked up through Vimeo's oEmbed API at build time and served from this site. A remote fetch that fails only costs the poster, never the build. The play button and wordmark are painted by CSS over the link — style.css, "video shortcodes" — so the markup is an anchor, an image and the button. @param {string} [0|id] the Vimeo video id @param {string} [1|image] optional local poster image; omit it to fetch the thumbnail from Vimeo at build time */ -}} {{- $id := "" -}} {{- $image := "" -}} {{- if .IsNamedParams -}} {{- $id = .Get "id" -}} {{- $image = .Get "image" -}} {{- else -}} {{- $id = .Get 0 -}} {{- $image = .Get 1 -}} {{- end -}} {{- /* The poster: a processed resource when one can be had, the raw path otherwise, an empty src only when a remote-only call is offline. */ -}} {{- $poster := "" -}} {{- $src := $image -}} {{- if $image -}} {{- $poster = resources.GetMatch $image -}} {{- else -}} {{- with try (resources.GetRemote (print "https://vimeo.com/api/oembed.json?width=1280&url=https%3A%2F%2Fvimeo.com%2F" $id)) -}} {{- with .Err -}} {{- warnf "vimeo shortcode: %s" . -}} {{- else -}} {{- with (unmarshal .Value.Content).thumbnail_url -}} {{- with try (resources.GetRemote .) -}} {{- with .Err -}} {{- warnf "vimeo shortcode: %s" . -}} {{- else -}} {{- $poster = .Value -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $poster -}} {{- $src = (.Fill "1200x675 webp Center q50").RelPermalink -}} {{- end -}}
