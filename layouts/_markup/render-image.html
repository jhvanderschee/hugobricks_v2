{{- /* Image render hook. The counterpart to `render-link.html`: a plain `` with the alt text the markdown gave it, and no `class` or `id`. Those come from a kramdown-style attribute list written after the image: ![A red paraglider over the bay](/photo.jpg){:.background} See `inline_attributes.html`, which hoists that list into this tag. Local raster images are served as resized webp (https://hugocodex.org/blog/ resize-all-images/): capped at 1920px wide — wide enough for the full-bleed `{:.background}` headers — and never upscaled. SVGs and remote URLs pass through untouched. @context {image} . @returns {string} one */ -}} {{- $src := .Destination -}} {{- $dims := "" -}} {{- with resources.GetMatch $src -}} {{- if ne .MediaType.SubType "svg" -}} {{- $img := . -}} {{- if gt .Width 1920 -}}{{- $img = .Resize "1920x webp q50" -}}{{- else -}}{{- $img = .Process "webp q50" -}}{{- end -}} {{- $src = $img.RelPermalink -}} {{- $dims = printf ` width="%d" height="%d"` $img.Width $img.Height -}} {{- end -}} {{- end -}} {{- /* no trailing newline: a render hook writes inline content */ -}}
